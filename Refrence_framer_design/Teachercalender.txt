type EventType = "class" | "assignment" | "test" | "ptm" | "reminder"

interface CalendarEvent {
    id: string
    title: string
    type: EventType
    date: Date
    startTime: string
    endTime: string
    location?: string
    mode?: "online" | "offline"
    description?: string
}

interface TeacherCalendarProps {
    primaryColor?: string
    backgroundColor?: string
}

type ViewMode = "month" | "week" | "day"

const eventColors: Record<EventType, string> = {
    class: "#2D5BFF",
    assignment: "#10B981",
    test: "#8B5CF6",
    ptm: "#F97316",
    reminder: "#6B7280",
}

const timeToMinutes = (time: string): number => {
    const [timePart, period] = time.split(" ")
    let [hours, minutes] = timePart.split(":").map(Number)
    if (period === "PM" && hours !== 12) hours += 12
    if (period === "AM" && hours === 12) hours = 0
    return hours * 60 + minutes
}

    const [currentDate, setCurrentDate] = useState(new Date())

    const [showAddEventSheet, setShowAddEventSheet] = useState(false)
    const [showMonthPicker, setShowMonthPicker] = useState(false)

            return events.filter(

                    e.date.getDate() === date.getDate() &&

                    e.date.getFullYear() === date.getFullYear()

        "January",

        "April",

        "June",

        "September",

        "December",

        new Date(date.getFullYear(), date.getMonth() + 1, 0).getDate()
    const getFirstDayOfMonth = (date: Date) => {
        const firstDay = new Date(
            date.getFullYear(),
            date.getMonth(),
            1
        ).getDay()
        return firstDay === 0 ? 6 : firstDay - 1

        const daysInMonth = getDaysInMonth(currentDate)
        const firstDay = getFirstDayOfMonth(currentDate)
        const days: (number | null)[] = []
        for (let i = 0; i < firstDay; i++) days.push(null)
        for (let i = 1; i <= daysInMonth; i++) days.push(i)
        return days
    }, [currentDate])

    const hours = useMemo(() => {
        const result: string[] = []
        for (let i = 6; i <= 22; i++) {
            const period = i >= 12 ? "PM" : "AM"
            const hour = i > 12 ? i - 12 : i === 0 ? 12 : i
            result.push(`${hour}:00 ${period}`)
        }
        return result
    }, [])

        const curr = new Date(currentDate)
        const first = curr.getDate() - curr.getDay() + 1
        const dates: Date[] = []
        for (let i = 0; i < 7; i++)
            dates.push(new Date(curr.setDate(first + i)))
        return dates
    }, [currentDate])

    const navigateMonth = (dir: "prev" | "next") => {
        setCurrentDate((p) => {
            const n = new Date(p)
            n.setMonth(n.getMonth() + (dir === "prev" ? -1 : 1))
            return n
        })
    }

    const navigateWeek = (dir: "prev" | "next") => {
        setCurrentDate((p) => {
            const n = new Date(p)
            n.setDate(n.getDate() + (dir === "prev" ? -7 : 7))
            return n
        })
    }

        setCurrentDate((p) => {
            const n = new Date(p)
            n.setDate(n.getDate() + (dir === "prev" ? -1 : 1))
            return n
        })
    }

        const today = new Date()
        return (
            day === today.getDate() &&
            currentDate.getMonth() === today.getMonth() &&
            currentDate.getFullYear() === today.getFullYear()
        )
    }

    const isSameDay = (d1: Date, d2: Date) =>
        d1.getDate() === d2.getDate() &&
        d1.getMonth() === d2.getMonth() &&
        d1.getFullYear() === d2.getFullYear()

    const isSelectedDay = (day: number) => {
        if (!selectedDate) return false
        return (
            day === selectedDate.getDate() &&
            currentDate.getMonth() === selectedDate.getMonth() &&
            currentDate.getFullYear() === selectedDate.getFullYear()
        )
    }

        const date = new Date(
            currentDate.getFullYear(),
            currentDate.getMonth(),
            day
        )
        setSelectedDate(date)
        if (getEventsForDate(date).length > 0) setShowEventSheet(true)

                return <Icons.Class />

                return <Icons.Assignment />

                return <Icons.Test />

                return <Icons.PTM />

                return <Icons.Clock />
        }

        const height = ((endMin - startMin) / 60) * 60
        return { top: `${top}px`, height: `${height}px` }
    }

    const getCurrentTimePosition = () => {

    }

                <AnimatePresence mode="wait">

                                                }}

                                                    style={{

                                                        fontWeight:

                                                            isSelected
                                                                ? 700

                                                            : isTodayDate

                                                              : "#111827",

                                                            display: "flex",
