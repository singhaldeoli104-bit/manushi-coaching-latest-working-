// Signup/Create Account Screen - Multi-step with validation, animations, and all states
// Step 1: Account Info | Step 2: Personal Details
import { addPropertyControls, ControlType } from "framer"
import { motion, useAnimation, AnimatePresence } from "framer-motion"
import { useState, useCallback, useMemo, useEffect } from "react"

interface SignupScreenProps {
    primaryColor: string
    backgroundColor: string
    onSignup?: (data: any) => Promise<void>
    onLogin?: () => void
    style?: any
}

const Icons = {
    Back: () => (
        <svg
            width="16"
            height="16"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <path d="M19 12H5M12 19l-7-7 7-7" />
        </svg>
    ),
    Check: () => (
        <svg
            width="16"
            height="16"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <polyline points="20 6 9 17 4 12" />
        </svg>
    ),
    Eye: () => (
        <svg
            width="20"
            height="20"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <path d="M1 12s4-8 11-8 11 8 11 8-4 8-11 8-11-8-11-8z" />
            <circle cx="12" cy="12" r="3" />
        </svg>
    ),
    EyeOff: () => (
        <svg
            width="20"
            height="20"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <path d="M17.94 17.94A10.07 10.07 0 0 1 12 20c-7 0-11-8-11-8a18.45 18.45 0 0 1 5.06-5.94M9.9 4.24A9.12 9.12 0 0 1 12 4c7 0 11 8 11 8a18.5 18.5 0 0 1-2.16 3.19m-6.72-1.07a3 3 0 1 1-4.24-4.24" />
            <line x1="1" y1="1" x2="23" y2="23" />
        </svg>
    ),
    Loader: () => (
        <svg
            width="20"
            height="20"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <path d="M12 2v4m0 12v4M4.93 4.93l2.83 2.83m8.48 8.48l2.83 2.83M2 12h4m12 0h4M4.93 19.07l2.83-2.83m8.48-8.48l2.83-2.83" />
        </svg>
    ),
    WifiOff: () => (
        <svg
            width="16"
            height="16"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <line x1="1" y1="1" x2="23" y2="23" />
            <path d="M16.72 11.06A10.94 10.94 0 0 1 19 12.55" />
        </svg>
    ),
    ChevronDown: () => (
        <svg
            width="16"
            height="16"
            viewBox="0 0 24 24"
            fill="none"
            stroke="currentColor"
            strokeWidth="2"
        >
            <polyline points="6 9 12 15 18 9" />
        </svg>
    ),
}

const subjects = [
    "English",
    "Science",
    "Math",
    "Physics",
    "Chemistry",
    "Biology",
    "Commerce",
    "Social Science",
    "Computer Science",
    "History",
    "Geography",
]
const roles = ["Teacher", "Teaching Assistant", "Mentor", "Coordinator"]
const grades = [
    "Grade 1-5",
    "Grade 6-8",
    "Grade 9-10",
    "Grade 11-12",
    "JEE",
    "NEET",
]
const experiences = [
    "0-1 years",
    "1-3 years",
    "3-5 years",
    "5-10 years",
    "10+ years",
]

/**
 * @framerSupportedLayoutWidth fixed
 * @framerSupportedLayoutHeight fixed
 */
export default function SignupScreen(props: SignupScreenProps) {
    const {
        primaryColor = "#2D5BFF",
        backgroundColor = "#F7F7F7",
        onSignup,
        onLogin,
    } = props

    const [step, setStep] = useState(1)
    const [isOnline, setIsOnline] = useState(true)
    const [isLoading, setIsLoading] = useState(false)

    // Step 1 fields
    const [emailPhone, setEmailPhone] = useState("")
    const [password, setPassword] = useState("")
    const [confirmPassword, setConfirmPassword] = useState("")
    const [showPassword, setShowPassword] = useState(false)
    const [showConfirmPassword, setShowConfirmPassword] = useState(false)
    const [agreedToTerms, setAgreedToTerms] = useState(false)

    // Step 2 fields
    const [fullName, setFullName] = useState("")
    const [role, setRole] = useState("Teacher")
    const [selectedSubjects, setSelectedSubjects] = useState<string[]>([])
    const [selectedGrades, setSelectedGrades] = useState<string[]>([])
    const [experience, setExperience] = useState("0-1 years")
    const [language, setLanguage] = useState("English")
    const [promoCode, setPromoCode] = useState("")

    // Focus states
    const [emailFocused, setEmailFocused] = useState(false)
    const [passwordFocused, setPasswordFocused] = useState(false)
    const [confirmPasswordFocused, setConfirmPasswordFocused] = useState(false)
    const [nameFocused, setNameFocused] = useState(false)

    const formControls = useAnimation()

    useEffect(() => {
        if (typeof window !== "undefined") {
            const handleOnline = () => setIsOnline(true)
            const handleOffline = () => setIsOnline(false)
            setIsOnline(navigator.onLine)
            window.addEventListener("online", handleOnline)
            window.addEventListener("offline", handleOffline)
            return () => {
                window.removeEventListener("online", handleOnline)
                window.removeEventListener("offline", handleOffline)
            }
        }
    }, [])

    const validateEmail = (email: string) =>
        /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
    const validatePhone = (phone: string) => /^\+?[\d\s-]{10,}$/.test(phone)

    const isEmailPhoneValid = useMemo(() => {
        return validateEmail(emailPhone) || validatePhone(emailPhone)
    }, [emailPhone])

    const passwordStrength = useMemo(() => {
        let strength = 0
        if (password.length >= 8) strength++
        if (/[A-Z]/.test(password)) strength++
        if (/[0-9]/.test(password)) strength++
        if (/[^A-Za-z0-9]/.test(password)) strength++
        return strength
    }, [password])

    const passwordsMatch =
        password === confirmPassword && confirmPassword.length > 0

    const isStep1Valid = useMemo(() => {
        return (
            isEmailPhoneValid &&
            password.length >= 8 &&
            passwordsMatch &&
            agreedToTerms
        )
    }, [isEmailPhoneValid, password, passwordsMatch, agreedToTerms])

    const isStep2Valid = useMemo(() => {
        return (
            fullName.length > 0 &&
            selectedSubjects.length > 0 &&
            selectedGrades.length > 0
        )
    }, [fullName, selectedSubjects, selectedGrades])

    const handleNext = useCallback(() => {
        if (step === 1 && isStep1Valid) {
            setStep(2)
        }
    }, [step, isStep1Valid])

    const handleBack = useCallback(() => {
        if (step === 2) {
            setStep(1)
        } else if (onLogin) {
            onLogin()
        }
    }, [step, onLogin])

    const handleSignup = useCallback(async () => {
        if (!isStep2Valid || !isOnline || isLoading) return

        setIsLoading(true)

        try {
            const data = {
                emailPhone,
                password,
                fullName,
                role,
                subjects: selectedSubjects,
                grades: selectedGrades,
                experience,
                language,
                promoCode,
            }
            if (onSignup) {
                await onSignup(data)
            } else {
                await new Promise((resolve) => setTimeout(resolve, 1500))
            }
        } catch (err) {
            formControls.start({
                x: [0, -10, 10, -10, 10, 0],
                transition: { duration: 0.4 },
            })
        } finally {
            setIsLoading(false)
        }
    }, [
        isStep2Valid,
        isOnline,
        isLoading,
        emailPhone,
        password,
        fullName,
        role,
        selectedSubjects,
        selectedGrades,
        experience,
        language,
        promoCode,
        onSignup,
        formControls,
    ])

    const toggleSubject = (subject: string) => {
        setSelectedSubjects((prev) =>
            prev.includes(subject)
                ? prev.filter((s) => s !== subject)
                : [...prev, subject]
        )
    }

    const toggleGrade = (grade: string) => {
        setSelectedGrades((prev) =>
            prev.includes(grade)
                ? prev.filter((g) => g !== grade)
                : [...prev, grade]
        )
    }

    const getPasswordStrengthColor = () => {
        if (passwordStrength === 0) return "#E5E7EB"
        if (passwordStrength <= 2) return "#EF4444"
        if (passwordStrength === 3) return "#F59E0B"
        return "#10B981"
    }

    const getPasswordStrengthText = () => {
        if (passwordStrength === 0) return ""
        if (passwordStrength <= 2) return "Weak"
        if (passwordStrength === 3) return "Medium"
        return "Strong"
    }

    return (
        <div
            style={{
                width: 390,
                height: 844,
                background: backgroundColor,
                fontFamily: "Inter, -apple-system, sans-serif",
                overflow: "hidden",
                position: "relative",
            }}
        >
            {!isOnline && (
                <motion.div
                    initial={{ y: -50, opacity: 0 }}
                    animate={{ y: 0, opacity: 1 }}
                    style={{
                        position: "absolute",
                        top: 0,
                        left: 0,
                        right: 0,
                        background: "#FEF3C7",
                        color: "#92400E",
                        padding: "8px 16px",
                        fontSize: 13,
                        fontWeight: 500,
                        display: "flex",
                        alignItems: "center",
                        gap: 8,
                        zIndex: 100,
                    }}
                >
                    <Icons.WifiOff />
                    No internet connection
                </motion.div>
            )}

            <div
                style={{
                    height: "100%",
                    display: "flex",
                    flexDirection: "column",
                }}
            >
                {/* Progress Bar */}
                <div style={{ padding: "24px 24px 16px" }}>
                    <div style={{ display: "flex", gap: 8 }}>
                        <motion.div
                            animate={{ background: primaryColor }}
                            style={{ flex: 1, height: 4, borderRadius: 2 }}
                        />
                        <motion.div
                            animate={{
                                background:
                                    step === 2 ? primaryColor : "#DADCE0",
                            }}
                            style={{ flex: 1, height: 4, borderRadius: 2 }}
                            transition={{ duration: 0.3 }}
                        />
                    </div>
                    <div
                        style={{
                            marginTop: 8,
                            fontSize: 12,
                            color: "#6B7280",
                            textAlign: "center",
                        }}
                    >
                        Step {step} of 2:{" "}
                        {step === 1
                            ? "Account Information"
                            : "Personal Details"}
                    </div>
                </div>

                {/* Content */}
                <div
                    style={{
                        flex: 1,
                        overflowY: "auto",
                        padding: "0 24px 24px",
                    }}
                >
                    <AnimatePresence mode="wait">
                        {step === 1 ? (
                            <motion.div
                                key="step1"
                                initial={{ opacity: 0, x: -20 }}
                                animate={{ opacity: 1, x: 0 }}
                                exit={{ opacity: 0, x: 20 }}
                                transition={{ duration: 0.3 }}
                            >
                                {/* Email/Phone */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Email or Phone Number
                                    </label>
                                    <motion.div
                                        animate={{
                                            borderColor: emailFocused
                                                ? primaryColor
                                                : "#DADCE0",
                                        }}
                                        style={{
                                            position: "relative",
                                            borderRadius: 8,
                                            border: "1px solid",
                                            transition: "border-color 0.2s",
                                        }}
                                    >
                                        <input
                                            type="text"
                                            placeholder="Enter your email or phone"
                                            value={emailPhone}
                                            onChange={(e) =>
                                                setEmailPhone(e.target.value)
                                            }
                                            onFocus={() =>
                                                setEmailFocused(true)
                                            }
                                            onBlur={() =>
                                                setEmailFocused(false)
                                            }
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: isEmailPhoneValid
                                                    ? 40
                                                    : 12,
                                                fontSize: 15,
                                                border: "none",
                                                outline: "none",
                                                background: "transparent",
                                                borderRadius: 8,
                                            }}
                                        />
                                        {isEmailPhoneValid && (
                                            <div
                                                style={{
                                                    position: "absolute",
                                                    right: 12,
                                                    top: "50%",
                                                    transform:
                                                        "translateY(-50%)",
                                                    color: "#10B981",
                                                }}
                                            >
                                                <Icons.Check />
                                            </div>
                                        )}
                                    </motion.div>
                                </div>

                                {/* Password */}
                                <div style={{ marginBottom: 12 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Password
                                    </label>
                                    <motion.div
                                        animate={{
                                            borderColor: passwordFocused
                                                ? primaryColor
                                                : "#DADCE0",
                                        }}
                                        style={{
                                            position: "relative",
                                            borderRadius: 8,
                                            border: "1px solid",
                                        }}
                                    >
                                        <input
                                            type={
                                                showPassword
                                                    ? "text"
                                                    : "password"
                                            }
                                            placeholder="Create a password"
                                            value={password}
                                            onChange={(e) =>
                                                setPassword(e.target.value)
                                            }
                                            onFocus={() =>
                                                setPasswordFocused(true)
                                            }
                                            onBlur={() =>
                                                setPasswordFocused(false)
                                            }
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: 44,
                                                fontSize: 15,
                                                border: "none",
                                                outline: "none",
                                                background: "transparent",
                                                borderRadius: 8,
                                            }}
                                        />
                                        <motion.button
                                            whileTap={{ scale: 0.9 }}
                                            onClick={() =>
                                                setShowPassword(!showPassword)
                                            }
                                            style={{
                                                position: "absolute",
                                                right: 12,
                                                top: "50%",
                                                transform: "translateY(-50%)",
                                                background: "transparent",
                                                border: "none",
                                                cursor: "pointer",
                                                padding: 4,
                                                color: "#9CA3AF",
                                            }}
                                        >
                                            {showPassword ? (
                                                <Icons.EyeOff />
                                            ) : (
                                                <Icons.Eye />
                                            )}
                                        </motion.button>
                                    </motion.div>
                                    {password.length > 0 && (
                                        <div style={{ marginTop: 8 }}>
                                            <div
                                                style={{
                                                    display: "flex",
                                                    gap: 4,
                                                    marginBottom: 4,
                                                }}
                                            >
                                                {[1, 2, 3, 4].map((i) => (
                                                    <motion.div
                                                        key={i}
                                                        animate={{
                                                            background:
                                                                i <=
                                                                passwordStrength
                                                                    ? getPasswordStrengthColor()
                                                                    : "#E5E7EB",
                                                        }}
                                                        style={{
                                                            flex: 1,
                                                            height: 4,
                                                            borderRadius: 2,
                                                        }}
                                                        transition={{
                                                            duration: 0.3,
                                                        }}
                                                    />
                                                ))}
                                            </div>
                                            <span
                                                style={{
                                                    fontSize: 12,
                                                    color: getPasswordStrengthColor(),
                                                    fontWeight: 500,
                                                }}
                                            >
                                                {getPasswordStrengthText()}
                                            </span>
                                        </div>
                                    )}
                                </div>

                                {/* Confirm Password */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Confirm Password
                                    </label>
                                    <motion.div
                                        animate={{
                                            borderColor: confirmPasswordFocused
                                                ? primaryColor
                                                : confirmPassword &&
                                                    !passwordsMatch
                                                  ? "#EF4444"
                                                  : "#DADCE0",
                                        }}
                                        style={{
                                            position: "relative",
                                            borderRadius: 8,
                                            border: "1px solid",
                                        }}
                                    >
                                        <input
                                            type={
                                                showConfirmPassword
                                                    ? "text"
                                                    : "password"
                                            }
                                            placeholder="Re-enter your password"
                                            value={confirmPassword}
                                            onChange={(e) =>
                                                setConfirmPassword(
                                                    e.target.value
                                                )
                                            }
                                            onFocus={() =>
                                                setConfirmPasswordFocused(true)
                                            }
                                            onBlur={() =>
                                                setConfirmPasswordFocused(false)
                                            }
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: 44,
                                                fontSize: 15,
                                                border: "none",
                                                outline: "none",
                                                background: "transparent",
                                                borderRadius: 8,
                                            }}
                                        />
                                        <motion.button
                                            whileTap={{ scale: 0.9 }}
                                            onClick={() =>
                                                setShowConfirmPassword(
                                                    !showConfirmPassword
                                                )
                                            }
                                            style={{
                                                position: "absolute",
                                                right: 12,
                                                top: "50%",
                                                transform: "translateY(-50%)",
                                                background: "transparent",
                                                border: "none",
                                                cursor: "pointer",
                                                padding: 4,
                                                color: "#9CA3AF",
                                            }}
                                        >
                                            {showConfirmPassword ? (
                                                <Icons.EyeOff />
                                            ) : (
                                                <Icons.Eye />
                                            )}
                                        </motion.button>
                                    </motion.div>
                                    {confirmPassword && !passwordsMatch && (
                                        <div
                                            style={{
                                                fontSize: 12,
                                                color: "#EF4444",
                                                marginTop: 4,
                                            }}
                                        >
                                            Passwords do not match
                                        </div>
                                    )}
                                </div>

                                {/* Terms */}
                                <label
                                    style={{
                                        display: "flex",
                                        alignItems: "start",
                                        gap: 12,
                                        cursor: "pointer",
                                        marginBottom: 32,
                                    }}
                                >
                                    <div
                                        style={{
                                            position: "relative",
                                            width: 44,
                                            height: 44,
                                            display: "flex",
                                            alignItems: "center",
                                            justifyContent: "center",
                                        }}
                                    >
                                        <input
                                            type="checkbox"
                                            checked={agreedToTerms}
                                            onChange={(e) =>
                                                setAgreedToTerms(
                                                    e.target.checked
                                                )
                                            }
                                            style={{
                                                width: 18,
                                                height: 18,
                                                cursor: "pointer",
                                            }}
                                        />
                                    </div>
                                    <span
                                        style={{
                                            fontSize: 14,
                                            color: "#374151",
                                            lineHeight: "44px",
                                        }}
                                    >
                                        I agree to the{" "}
                                        <span
                                            style={{
                                                color: primaryColor,
                                                fontWeight: 500,
                                            }}
                                        >
                                            Terms & Privacy Policy
                                        </span>
                                    </span>
                                </label>

                                {/* Next Button */}
                                <motion.button
                                    whileHover={
                                        isStep1Valid ? { scale: 1.02 } : {}
                                    }
                                    whileTap={
                                        isStep1Valid ? { scale: 0.98 } : {}
                                    }
                                    onClick={handleNext}
                                    disabled={!isStep1Valid}
                                    style={{
                                        width: "100%",
                                        height: 52,
                                        borderRadius: 12,
                                        border: "none",
                                        background: isStep1Valid
                                            ? primaryColor
                                            : "#E0E0E0",
                                        color: isStep1Valid
                                            ? "#FFF"
                                            : "#9CA3AF",
                                        fontSize: 16,
                                        fontWeight: 600,
                                        cursor: isStep1Valid
                                            ? "pointer"
                                            : "not-allowed",
                                        marginBottom: 24,
                                    }}
                                >
                                    Next
                                </motion.button>

                                <div
                                    style={{
                                        textAlign: "center",
                                        fontSize: 14,
                                        color: "#6B7280",
                                    }}
                                >
                                    Already have an account?{" "}
                                    <motion.button
                                        whileHover={{
                                            textDecoration: "underline",
                                        }}
                                        onClick={onLogin}
                                        style={{
                                            background: "transparent",
                                            border: "none",
                                            color: primaryColor,
                                            fontWeight: 600,
                                            cursor: "pointer",
                                            padding: 0,
                                        }}
                                    >
                                        Log in
                                    </motion.button>
                                </div>
                            </motion.div>
                        ) : (
                            <motion.div
                                key="step2"
                                initial={{ opacity: 0, x: 20 }}
                                animate={{ opacity: 1, x: 0 }}
                                exit={{ opacity: 0, x: -20 }}
                                transition={{ duration: 0.3 }}
                            >
                                {/* Full Name */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Full Name
                                    </label>
                                    <motion.div
                                        animate={{
                                            borderColor: nameFocused
                                                ? primaryColor
                                                : "#DADCE0",
                                        }}
                                        style={{
                                            borderRadius: 8,
                                            border: "1px solid",
                                        }}
                                    >
                                        <input
                                            type="text"
                                            placeholder="Enter your full name"
                                            value={fullName}
                                            onChange={(e) =>
                                                setFullName(
                                                    e.target.value.replace(
                                                        /[0-9]/g,
                                                        ""
                                                    )
                                                )
                                            }
                                            onFocus={() => setNameFocused(true)}
                                            onBlur={() => setNameFocused(false)}
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: 12,
                                                fontSize: 15,
                                                border: "none",
                                                outline: "none",
                                                background: "transparent",
                                                borderRadius: 8,
                                            }}
                                        />
                                    </motion.div>
                                </div>

                                {/* Role */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Role
                                    </label>
                                    <div style={{ position: "relative" }}>
                                        <select
                                            value={role}
                                            onChange={(e) =>
                                                setRole(e.target.value)
                                            }
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: 36,
                                                fontSize: 15,
                                                border: "1px solid #DADCE0",
                                                borderRadius: 8,
                                                background: "#FFF",
                                                outline: "none",
                                                cursor: "pointer",
                                                appearance: "none",
                                            }}
                                        >
                                            {roles.map((r) => (
                                                <option key={r} value={r}>
                                                    {r}
                                                </option>
                                            ))}
                                        </select>
                                        <div
                                            style={{
                                                position: "absolute",
                                                right: 12,
                                                top: "50%",
                                                transform: "translateY(-50%)",
                                                pointerEvents: "none",
                                            }}
                                        >
                                            <Icons.ChevronDown />
                                        </div>
                                    </div>
                                </div>

                                {/* Subjects */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Subjects You Teach
                                    </label>
                                    <div
                                        style={{
                                            display: "flex",
                                            flexWrap: "wrap",
                                            gap: 8,
                                        }}
                                    >
                                        {subjects.map((subject) => (
                                            <motion.button
                                                key={subject}
                                                whileTap={{ scale: 0.95 }}
                                                onClick={() =>
                                                    toggleSubject(subject)
                                                }
                                                style={{
                                                    padding: "8px 16px",
                                                    borderRadius: 20,
                                                    border: "1px solid",
                                                    borderColor:
                                                        selectedSubjects.includes(
                                                            subject
                                                        )
                                                            ? primaryColor
                                                            : "#DADCE0",
                                                    background:
                                                        selectedSubjects.includes(
                                                            subject
                                                        )
                                                            ? primaryColor
                                                            : "#FFF",
                                                    color: selectedSubjects.includes(
                                                        subject
                                                    )
                                                        ? "#FFF"
                                                        : "#374151",
                                                    fontSize: 14,
                                                    fontWeight: 500,
                                                    cursor: "pointer",
                                                    display: "flex",
                                                    alignItems: "center",
                                                    gap: 6,
                                                }}
                                            >
                                                {selectedSubjects.includes(
                                                    subject
                                                ) && <Icons.Check />}
                                                {subject}
                                            </motion.button>
                                        ))}
                                    </div>
                                </div>

                                {/* Grades */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Grade Levels
                                    </label>
                                    <div
                                        style={{
                                            display: "flex",
                                            flexWrap: "wrap",
                                            gap: 8,
                                        }}
                                    >
                                        {grades.map((grade) => (
                                            <motion.button
                                                key={grade}
                                                whileTap={{ scale: 0.95 }}
                                                onClick={() =>
                                                    toggleGrade(grade)
                                                }
                                                style={{
                                                    padding: "8px 16px",
                                                    borderRadius: 20,
                                                    border: "1px solid",
                                                    borderColor:
                                                        selectedGrades.includes(
                                                            grade
                                                        )
                                                            ? primaryColor
                                                            : "#DADCE0",
                                                    background:
                                                        selectedGrades.includes(
                                                            grade
                                                        )
                                                            ? primaryColor
                                                            : "#FFF",
                                                    color: selectedGrades.includes(
                                                        grade
                                                    )
                                                        ? "#FFF"
                                                        : "#374151",
                                                    fontSize: 14,
                                                    fontWeight: 500,
                                                    cursor: "pointer",
                                                    display: "flex",
                                                    alignItems: "center",
                                                    gap: 6,
                                                }}
                                            >
                                                {selectedGrades.includes(
                                                    grade
                                                ) && <Icons.Check />}
                                                {grade}
                                            </motion.button>
                                        ))}
                                    </div>
                                </div>

                                {/* Experience */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Years of Experience
                                    </label>
                                    <div style={{ position: "relative" }}>
                                        <select
                                            value={experience}
                                            onChange={(e) =>
                                                setExperience(e.target.value)
                                            }
                                            style={{
                                                width: "100%",
                                                height: 48,
                                                paddingLeft: 12,
                                                paddingRight: 36,
                                                fontSize: 15,
                                                border: "1px solid #DADCE0",
                                                borderRadius: 8,
                                                background: "#FFF",
                                                outline: "none",
                                                cursor: "pointer",
                                                appearance: "none",
                                            }}
                                        >
                                            {experiences.map((exp) => (
                                                <option key={exp} value={exp}>
                                                    {exp}
                                                </option>
                                            ))}
                                        </select>
                                        <div
                                            style={{
                                                position: "absolute",
                                                right: 12,
                                                top: "50%",
                                                transform: "translateY(-50%)",
                                                pointerEvents: "none",
                                            }}
                                        >
                                            <Icons.ChevronDown />
                                        </div>
                                    </div>
                                </div>

                                {/* Language */}
                                <div style={{ marginBottom: 20 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Language Preference
                                    </label>
                                    <div style={{ display: "flex", gap: 12 }}>
                                        {["English", "Hindi", "Both"].map(
                                            (lang) => (
                                                <motion.button
                                                    key={lang}
                                                    whileTap={{ scale: 0.95 }}
                                                    onClick={() =>
                                                        setLanguage(lang)
                                                    }
                                                    style={{
                                                        flex: 1,
                                                        height: 48,
                                                        borderRadius: 8,
                                                        border: "1px solid",
                                                        borderColor:
                                                            language === lang
                                                                ? primaryColor
                                                                : "#DADCE0",
                                                        background:
                                                            language === lang
                                                                ? `${primaryColor}15`
                                                                : "#FFF",
                                                        color:
                                                            language === lang
                                                                ? primaryColor
                                                                : "#374151",
                                                        fontSize: 14,
                                                        fontWeight: 500,
                                                        cursor: "pointer",
                                                    }}
                                                >
                                                    {lang}
                                                </motion.button>
                                            )
                                        )}
                                    </div>
                                </div>

                                {/* Promo Code */}
                                <div style={{ marginBottom: 32 }}>
                                    <label
                                        style={{
                                            fontSize: 14,
                                            fontWeight: 600,
                                            color: "#374151",
                                            display: "block",
                                            marginBottom: 8,
                                        }}
                                    >
                                        Promotional Code (Optional)
                                    </label>
                                    <input
                                        type="text"
                                        placeholder="Enter code"
                                        value={promoCode}
                                        onChange={(e) =>
                                            setPromoCode(
                                                e.target.value.toUpperCase()
                                            )
                                        }
                                        style={{
                                            width: "100%",
                                            height: 48,
                                            paddingLeft: 12,
                                            paddingRight: 12,
                                            fontSize: 15,
                                            border: "1px solid #DADCE0",
                                            borderRadius: 8,
                                            outline: "none",
                                        }}
                                    />
                                </div>

                                {/* Create Account Button */}
                                <motion.button
                                    whileHover={
                                        isStep2Valid && isOnline && !isLoading
                                            ? { scale: 1.02 }
                                            : {}
                                    }
                                    whileTap={
                                        isStep2Valid && isOnline && !isLoading
                                            ? { scale: 0.98 }
                                            : {}
                                    }
                                    onClick={handleSignup}
                                    disabled={
                                        !isStep2Valid || !isOnline || isLoading
                                    }
                                    style={{
                                        width: "100%",
                                        height: 52,
                                        borderRadius: 12,
                                        border: "none",
                                        background:
                                            isStep2Valid && isOnline
                                                ? primaryColor
                                                : "#E0E0E0",
                                        color:
                                            isStep2Valid && isOnline
                                                ? "#FFF"
                                                : "#9CA3AF",
                                        fontSize: 16,
                                        fontWeight: 600,
                                        cursor:
                                            isStep2Valid &&
                                            !isLoading &&
                                            isOnline
                                                ? "pointer"
                                                : "not-allowed",
                                        display: "flex",
                                        alignItems: "center",
                                        justifyContent: "center",
                                        gap: 8,
                                    }}
                                >
                                    {isLoading ? (
                                        <>
                                            <motion.div
                                                animate={{ rotate: 360 }}
                                                transition={{
                                                    duration: 1,
                                                    repeat: Infinity,
                                                    ease: "linear",
                                                }}
                                            >
                                                <Icons.Loader />
                                            </motion.div>
                                            Creating Account...
                                        </>
                                    ) : (
                                        "Create Account"
                                    )}
                                </motion.button>
                            </motion.div>
                        )}
                    </AnimatePresence>
                </div>
            </div>
        </div>
    )
}
