interface ProfileSetupProps {
    primaryColor?: string
    backgroundColor?: string
    onContinue?: () => void
}

export default function ProfileSetup(props: ProfileSetupProps) {
    const {
        primaryColor = "#7C3AED",
        backgroundColor = "#F9FAFB",
        onContinue,
    } = props

    // Avatar state
    const [avatar, setAvatar] = useState<string | null>(null)
    const [showAvatarModal, setShowAvatarModal] = useState(false)

    // Personal details
    const [fullName, setFullName] = useState("")
    const [gender, setGender] = useState("")
    const [showGenderDropdown, setShowGenderDropdown] = useState(false)
    const [dob, setDob] = useState("")
    const [showDatePicker, setShowDatePicker] = useState(false)

    // Professional details
    const [role, setRole] = useState("")
    const [showRoleDropdown, setShowRoleDropdown] = useState(false)
    const [experience, setExperience] = useState("")
    const [showExperienceDropdown, setShowExperienceDropdown] = useState(false)
    const [qualification, setQualification] = useState("")
    const [showQualificationDropdown, setShowQualificationDropdown] =
        useState(false)

    // Subject expertise
    const [selectedSubjects, setSelectedSubjects] = useState<string[]>([])
    const [customSubject, setCustomSubject] = useState("")
    const [showCustomSubjectInput, setShowCustomSubjectInput] = useState(false)

    // Grade levels
    const [selectedGrades, setSelectedGrades] = useState<string[]>([])
    const [selectedCompetitiveExams, setSelectedCompetitiveExams] = useState<
        string[]
    >([])

    // Teaching medium
    const [selectedLanguages, setSelectedLanguages] = useState<string[]>([])
    const [isBilingual, setIsBilingual] = useState(false)

    // Validation states
    const [errors, setErrors] = useState<Record<string, string>>({})
    const [touched, setTouched] = useState<Record<string, boolean>>({})

    // Loading states
    const [isLoading, setIsLoading] = useState(false)
    const [isSaving, setIsSaving] = useState(false)

    // Dropdowns data
    const genderOptions = ["Male", "Female", "Other", "Prefer not to say"]
    const roleOptions = [
        "Teacher",
        "Assistant Teacher",
        "Mentor",
        "Academic Coordinator",
    ]
    const experienceOptions = [
        "0-2 years",
        "2-5 years",
        "5-10 years",
        "10-15 years",
        "15+ years",
    ]
    const qualificationOptions = [
        "Bachelor's Degree",
        "Master's Degree",
        "PhD",
        "Diploma",
        "Other",
    ]

    const subjectOptions = [
        "Mathematics",
        "Physics",
        "Chemistry",
        "Biology",
        "English",
        "Hindi",
        "Social Science",
        "Computer Science",
        "Commerce",
        "Economics",
        "Accountancy",
    ]

    const gradeOptions = [
        { label: "Primary (1-5)", value: "primary" },
        { label: "Middle (6-8)", value: "middle" },
        { label: "High School (9-10)", value: "high" },
        { label: "Senior Secondary (11-12)", value: "senior" },
    ]

    const competitiveExamOptions = [
        "JEE (Main/Advanced)",
        "NEET",
        "UPSC",
        "CAT/MBA",
        "Other Competitive Exams",
    ]

    const languageOptions = ["English", "Hindi", "Regional Language"]

    // Calculate age from DOB
    const calculateAge = useCallback((dateString: string): number | null => {
        if (!dateString || dateString.length !== 10) return null
        const [day, month, year] = dateString.split("/").map(Number)
        if (!day || !month || !year) return null

        const today = new Date()
        const birthDate = new Date(year, month - 1, day)
        let age = today.getFullYear() - birthDate.getFullYear()
        const monthDiff = today.getMonth() - birthDate.getMonth()

        if (
            monthDiff < 0 ||
            (monthDiff === 0 && today.getDate() < birthDate.getDate())
        ) {
            age--
        }

        return age
    }, [])

    // Validate DOB (must be ≥18)
    const validateDob = useCallback(
        (dateString: string): string => {
            const age = calculateAge(dateString)
            if (age === null) return "Invalid date format"
            if (age < 18) return "You must be at least 18 years old"
            if (age > 100) return "Please enter a valid date of birth"
            return ""
        },
        [calculateAge]
    )

    // Handle DOB input with auto-formatting
    const handleDobChange = (value: string) => {
        // Remove non-digits
        const cleaned = value.replace(/\D/g, "")

        // Format as dd/mm/yyyy
        let formatted = cleaned
        if (cleaned.length >= 2) {
            formatted = cleaned.slice(0, 2) + "/" + cleaned.slice(2)
        }
        if (cleaned.length >= 4) {
            formatted =
                cleaned.slice(0, 2) +
                "/" +
                cleaned.slice(2, 4) +
                "/" +
                cleaned.slice(4, 8)
        }

        setDob(formatted)

        // Validate if complete
        if (formatted.length === 10) {
            const error = validateDob(formatted)
            setErrors((prev) => ({ ...prev, dob: error }))
        }
    }

    // Avatar upload simulation
    const handleAvatarUpload = (source: "camera" | "gallery") => {
        setShowAvatarModal(false)
        // Simulate image upload
        setTimeout(() => {
            setAvatar("https://via.placeholder.com/120")
        }, 500)
    }

    // Toggle selections
    const toggleSubject = (subject: string) => {
        setSelectedSubjects((prev) =>
            prev.includes(subject)
                ? prev.filter((s) => s !== subject)
                : [...prev, subject]
        )
    }

    const toggleGrade = (grade: string) => {
        setSelectedGrades((prev) =>
            prev.includes(grade)
                ? prev.filter((g) => g !== grade)
                : [...prev, grade]
        )
    }

    const toggleCompetitiveExam = (exam: string) => {
        setSelectedCompetitiveExams((prev) =>
            prev.includes(exam)
                ? prev.filter((e) => e !== exam)
                : [...prev, exam]
        )
    }

    const toggleLanguage = (language: string) => {
        setSelectedLanguages((prev) =>
            prev.includes(language)
                ? prev.filter((l) => l !== language)
                : [...prev, language]
        )
    }

    // Add custom subject
    const handleAddCustomSubject = () => {
        if (
            customSubject.trim() &&
            !selectedSubjects.includes(customSubject.trim())
        ) {
            setSelectedSubjects((prev) => [...prev, customSubject.trim()])
            setCustomSubject("")
            setShowCustomSubjectInput(false)
        }
    }

    // Form validation
    const isFormValid = useMemo(() => {
        return (
            fullName.trim().length >= 2 &&
            gender !== "" &&
            dob.length === 10 &&
            !errors.dob &&
            role !== "" &&
            experience !== "" &&
            qualification !== "" &&
            selectedSubjects.length > 0 &&
            (selectedGrades.length > 0 ||
                selectedCompetitiveExams.length > 0) &&
            selectedLanguages.length > 0
        )
    }, [
        fullName,
        gender,
        dob,
        errors.dob,
        role,
        experience,
        qualification,
        selectedSubjects,
        selectedGrades,
        selectedCompetitiveExams,
        selectedLanguages,
    ])

    // Handle continue
    const handleContinue = async () => {
        if (!isFormValid) return

        setIsLoading(true)

        // Simulate API call
        await new Promise((resolve) => setTimeout(resolve, 1500))

        setIsLoading(false)
        onContinue?.()
    }

    // Auto-save draft every 10 seconds
    useEffect(() => {
        const interval = setInterval(() => {
            if (fullName || gender || dob || role) {
                setIsSaving(true)
                setTimeout(() => setIsSaving(false), 1000)
            }
        }, 10000)

        return () => clearInterval(interval)
    }, [fullName, gender, dob, role])

    return (
        <div
            style={{
                width: "390px",
                height: "844px",
                background: backgroundColor,
                fontFamily:
                    "-apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif",
                position: "relative",
                overflow: "auto",
                WebkitOverflowScrolling: "touch",
            }}
        >
            {/* Scrollable Content */}
            <div style={{ padding: "24px 20px 120px 20px" }}>
                {/* Section A: Profile Photo */}
                <div style={{ marginBottom: "32px" }}>
                    <h2
                        style={{
                            margin: "0 0 16px 0",
                            fontSize: "16px",
                            fontWeight: 600,
                            color: "#111827",
                        }}
                    >
                        Profile Photo
                    </h2>
                    <div style={{ display: "flex", justifyContent: "center" }}>
                        <motion.button
                            whileHover={{ scale: 1.02 }}
                            whileTap={{ scale: 0.98 }}
                            onClick={() => setShowAvatarModal(true)}
                            style={{
                                width: "120px",
                                height: "120px",
                                borderRadius: "50%",
                                border: avatar
                                    ? "none"
                                    : `3px dashed ${primaryColor}30`,
                                background: avatar
                                    ? `url(${avatar}) center/cover`
                                    : `${primaryColor}10`,
                                display: "flex",
                                alignItems: "center",
                                justifyContent: "center",
                                cursor: "pointer",
                                position: "relative",
                                overflow: "hidden",
                            }}
                            aria-label="Upload profile photo"
                        >
                            {!avatar && (
                                <div
                                    style={{
                                        textAlign: "center",
                                        color: primaryColor,
                                    }}
                                >
                                    <Icons.Camera />
                                    <div
                                        style={{
                                            fontSize: "12px",
                                            marginTop: "8px",
                                            fontWeight: 500,
                                        }}
                                    >
                                        Add Photo
                                    </div>
                                </div>
                            )}
                            {avatar && (
                                <motion.div
                                    initial={{ opacity: 0 }}
                                    animate={{ opacity: 1 }}
                                    style={{
                                        position: "absolute",
                                        bottom: "0",
                                        right: "0",
                                        background: primaryColor,
                                        width: "32px",
                                        height: "32px",
                                        borderRadius: "50%",
                                        display: "flex",
                                        alignItems: "center",
                                        justifyContent: "center",
                                        color: "#FFFFFF",
                                        border: "2px solid #FFFFFF",
                                    }}
                                >
                                    <Icons.Edit />
                                </motion.div>
                            )}
                        </motion.button>
                    </div>
                </div>

                {/* Section B: Personal Details */}
                <div style={{ marginBottom: "32px" }}>
                    <h2
                        style={{
                            margin: "0 0 16px 0",
                            fontSize: "16px",
                            fontWeight: 600,
                            color: "#111827",
                        }}
                    >
                        Personal Details
                    </h2>

                    {/* Full Name */}
                    <div style={{ marginBottom: "16px" }}>
                        <label
                            htmlFor="fullName"
                            style={{
                                display: "block",
                                fontSize: "14px",
                                fontWeight: 500,
                                color: "#374151",
                                marginBottom: "8px",
                            }}
                        >
                            Full Name{" "}
                            <span style={{ color: "#EF4444" }}>*</span>
                        </label>
                        <motion.input
                            whileFocus={{
                                boxShadow: `0 0 0 3px ${primaryColor}20`,
                            }}
                            id="fullName"
                            type="text"
                            value={fullName}
                            onChange={(
                                e: React.ChangeEvent<HTMLInputElement>
                            ) => {
                                // Filter out numbers
                                const filtered = e.target.value.replace(
                                    /[0-9]/g,
                                    ""
                                )
                                setFullName(filtered)
                            }}
                            placeholder="Enter your full name"
                            style={{
                                width: "100%",
                                padding: "12px 16px",
                                fontSize: "16px",
                                border: "1.5px solid #D1D5DB",
                                borderRadius: "12px",
                                outline: "none",
                                transition: "all 0.2s",
                                fontFamily: "inherit",
                            }}
                            aria-label="Full name"
                        />
                    </div>

                    {/* Gender */}
                    <div style={{ marginBottom: "16px" }}>
                        <label
                            style={{
                                display: "block",
                                fontSize: "14px",
                                fontWeight: 500,
                                color: "#374151",
                                marginBottom: "8px",
                            }}
                        >
                            Gender <span style={{ color: "#EF4444" }}>*</span>
                        </label>
                        <div style={{ position: "relative" }}>
                            <motion.button
                                whileTap={{ scale: 0.98 }}
                                onClick={() =>
                                    setShowGenderDropdown(!showGenderDropdown)
                                }
                                style={{
                                    width: "100%",
                                    padding: "12px 16px",
                                    fontSize: "16px",
                                    border: "1.5px solid #D1D5DB",
                                    borderRadius: "12px",
                                    background: "#FFFFFF",
                                    textAlign: "left",
                                    cursor: "pointer",
                                    display: "flex",
                                    alignItems: "center",
                                    justifyContent: "space-between",
                                    color: gender ? "#111827" : "#9CA3AF",
                                }}
                                aria-label="Select gender"
                            >
                                {gender || "Select gender"}
                                <Icons.ChevronDown />
                            </motion.button>

                            <AnimatePresence>
                                {showGenderDropdown && (
                                    <motion.div
                                        initial={{ opacity: 0, y: -10 }}
                                        animate={{ opacity: 1, y: 0 }}
                                        exit={{ opacity: 0, y: -10 }}
                                        style={{
                                            position: "absolute",
                                            top: "calc(100% + 8px)",
                                            left: 0,
                                            right: 0,
                                            background: "#FFFFFF",
                                            border: "1px solid #E5E7EB",
                                            borderRadius: "12px",
                                            boxShadow:
                                                "0 10px 25px rgba(0,0,0,0.1)",
                                            zIndex: 20,
                                            overflow: "hidden",
                                        }}
                                    >
                                        {genderOptions.map((option, index) => (
                                            <motion.button
                                                key={option}
                                                whileHover={{
                                                    background: `${primaryColor}10`,
                                                }}
                                                onClick={() => {
                                                    setGender(option)
                                                    setShowGenderDropdown(false)
                                                }}
                                                style={{
                                                    width: "100%",
                                                    padding: "12px 16px",
                                                    fontSize: "15px",
                                                    border: "none",
                                                    background:
                                                        gender === option
                                                            ? `${primaryColor}10`
                                                            : "transparent",
                                                    textAlign: "left",
                                                    cursor: "pointer",
                                                    color:
                                                        gender === option
                                                            ? primaryColor
                                                            : "#374151",
                                                    fontWeight:
                                                        gender === option
                                                            ? 500
                                                            : 400,
                                                    borderTop:
                                                        index > 0
                                                            ? "1px solid #F3F4F6"
                                                            : "none",
                                                }}
                                            >
                                                {option}
                                            </motion.button>
                                        ))}
                                    </motion.div>
                                )}
                            </AnimatePresence>
                        </div>
                    </div>

                    {/* Date of Birth */}
                    <div>
                        <label
                            htmlFor="dob"
                            style={{
                                display: "block",
                                fontSize: "14px",
                                fontWeight: 500,
                                color: "#374151",
                                marginBottom: "8px",
                            }}
                        >
                            Date of Birth{" "}
                            <span style={{ color: "#EF4444" }}>*</span>
                        </label>
                        <div style={{ position: "relative" }}>
                            <motion.input
                                whileFocus={{
                                    boxShadow: `0 0 0 3px ${primaryColor}20`,
                                }}
                                id="dob"
                                type="text"
                                value={dob}
                                onChange={(
                                    e: React.ChangeEvent<HTMLInputElement>
                                ) => handleDobChange(e.target.value)}
                                placeholder="dd/mm/yyyy"
                                maxLength={10}
                                style={{
                                    width: "100%",
                                    padding: "12px 44px 12px 16px",
                                    fontSize: "16px",
                                    border: `1.5px solid ${errors.dob ? "#EF4444" : "#D1D5DB"}`,
                                    borderRadius: "12px",
                                    outline: "none",
                                    transition: "all 0.2s",
                                    fontFamily: "inherit",
                                }}
                                aria-label="Date of birth"
                            />
                            <div
                                style={{
                                    position: "absolute",
                                    right: "12px",
                                    top: "50%",
                                    transform: "translateY(-50%)",
                                    color: "#9CA3AF",
                                    pointerEvents: "none",
                                }}
                            >
                                <Icons.Calendar />
                            </div>
                        </div>
                        {errors.dob && (
                            <motion.p
                                initial={{ opacity: 0, y: -5 }}
                                animate={{ opacity: 1, y: 0 }}
                                style={{
                                    margin: "6px 0 0 0",
                                    fontSize: "13px",
                                    color: "#EF4444",
                                }}
                            >
                                {errors.dob}
                            </motion.p>
                        )}
                        {dob.length === 10 && !errors.dob && (
                            <motion.p
                                initial={{ opacity: 0, y: -5 }}
                                animate={{ opacity: 1, y: 0 }}
                                style={{
                                    margin: "6px 0 0 0",
                                    fontSize: "13px",
                                    color: "#059669",
                                }}
                            >
                                Age: {calculateAge(dob)} years
                            </motion.p>
                        )}
                    </div>
                </div>

                {/* Remaining sections omitted for brevity - they stay the same */}
                {/* Professional Details, Subject Expertise, Grade Levels, Teaching Medium, etc. */}
                {/* ... (rest of the form sections remain unchanged) ... */}
            </div>

            {/* Fixed Bottom Button - stays the same */}
            <div
                style={{
                    position: "fixed",
                    bottom: 0,
                    left: 0,
                    right: 0,
                    padding: "16px 20px 24px 20px",
                    background:
                        "linear-gradient(to top, #FFFFFF 80%, transparent)",
                    borderTop: "1px solid #F3F4F6",
                }}
            >
                <motion.button
                    whileHover={{ scale: isFormValid ? 1.02 : 1 }}
                    whileTap={{ scale: isFormValid ? 0.98 : 1 }}
                    onClick={handleContinue}
                    disabled={!isFormValid || isLoading}
                    style={{
                        width: "100%",
                        padding: "16px",
                        fontSize: "16px",
                        fontWeight: 600,
                        color: "#FFFFFF",
                        background: isFormValid ? primaryColor : "#D1D5DB",
                        border: "none",
                        borderRadius: "14px",
                        cursor: isFormValid ? "pointer" : "not-allowed",
                        boxShadow: isFormValid
                            ? "0 4px 12px rgba(124, 58, 237, 0.3)"
                            : "none",
                        transition: "all 0.3s",
                        display: "flex",
                        alignItems: "center",
                        justifyContent: "center",
                        gap: "8px",
                    }}
                    aria-label="Save and continue"
                >
                    {isLoading ? (
                        <>
                            <motion.div
                                animate={{ rotate: 360 }}
                                transition={{
                                    duration: 1,
                                    repeat: Infinity,
                                    ease: "linear",
                                }}
                                style={{
                                    width: "20px",
                                    height: "20px",
                                    border: "2px solid #FFFFFF50",
                                    borderTopColor: "#FFFFFF",
                                    borderRadius: "50%",
                                }}
                            />
                            Saving...
                        </>
                    ) : (
                        "Save & Continue"
                    )}
                </motion.button>

                {!isFormValid && (
                    <motion.p
                        initial={{ opacity: 0 }}
                        animate={{ opacity: 1 }}
                        style={{
                            margin: "10px 0 0 0",
                            fontSize: "13px",
                            color: "#EF4444",
                            textAlign: "center",
                        }}
                    >
                        Please fill all required fields to continue
                    </motion.p>
                )}
            </div>

            {/* Avatar Upload Modal - stays the same */}
            <AnimatePresence>
                {showAvatarModal && (
                    <>
                        <motion.div
                            initial={{ opacity: 0 }}
                            animate={{ opacity: 1 }}
